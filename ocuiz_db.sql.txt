#CREATE DB
create database ociuz_db;

# connect db ociuz_db
\c ociuz_db; 

#CREATE TABLE - Users
create table users(user_id serial primary key,name varchar(255) not null,email varchar(255) unique not null,registration_date timestamp default current_timestamp);

#CREATE TABLE - Products
create table products(product_id serial primary key,name varchar(255) not null,category varchar(100) not null,price decimal(10,2) not null,stock int not null);

#CREATE TABLE - Transactions
create table transactions(transaction_id serial primary key,user_id int references users(user_id),date timestamp default current_timestamp);

#CREATE TABLE - Transaction Details
create table transactions_details(transaction_id int references transactions(transaction_id),product_id int references products(product_id),quantity int not null,primary key(transaction_id,product_id));

#Find users exceeding $500 in total purchases

#Function to Calculate Remaining Stock for a Product

create or replace function calculate_remaining_stock(product_id int)
return int as $$
declare
    remaining_stock int;
begin
    
    select stock - COALESCE(sum(td.quantity), 0)
    into remaining_stock
    FROM products p
    LEFT JOIN transactions_details td ON p.product_id = td.product_id
    WHERE p.product_id = $1 
    GROUP BY p.product_id;

    return remaining_stock;
end;
$$ language plpgsql;
